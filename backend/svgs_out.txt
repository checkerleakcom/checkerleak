    // Render badge HTML
    function renderBadge(badge) {
      if (!badge) return '';
      if (badge === 'HOT') {
        return `<span class="px-1.5 py-0.5 text-[9px] font-extrabold uppercase rounded bg-rose-500 text-white tracking-wider shrink-0 shadow-xs">HOT</span>`;
      } else if (badge === 'NEW') {
        return `<span class="px-1.5 py-0.5 text-[9px] font-extrabold uppercase rounded bg-purple-600 text-white tracking-wider shrink-0 shadow-xs">NEW</span>`;
      } else if (badge === 'PRO') {
        return `<span class="px-1.5 py-0.5 text-[9px] font-extrabold uppercase rounded bg-amber-500 text-white tracking-wider shrink-0 shadow-xs">PRO</span>`;
      } else if (badge === 'EXCLUSIVE') {
        return `<span class="px-1.5 py-0.5 text-[9px] font-extrabold uppercase rounded bg-emerald-600 text-white tracking-wider shrink-0 shadow-xs">EXCLUSIVE</span>`;
      }
      return '';
    }

    // Render the 90+ Services Directory
    function renderServicesDirectory() {
      const container = document.getElementById("servicesDirectoryContainer");
      if (!container) return;

      const q = currentDirSearch.trim().toLowerCase();
      let filtered = ALL_SERVICES.filter(s => {
        const matchesCat = (currentDirCategory === 'ALL') || (s.cat === currentDirCategory);
        const matchesQuery = !q || s.name.toLowerCase().includes(q) || s.catTitle.toLowerCase().includes(q) || s.desc.toLowerCase().includes(q);
        return matchesCat && matchesQuery;
      });

      // Update count badge
      const badgeEl = document.getElementById("serviceCountBadge");
      if (badgeEl) badgeEl.innerText = `${filtered.length} Checkers`;

      if (filtered.length === 0) {
        container.innerHTML = `
          <div class="bg-white rounded-2xl border border-slate-200 p-12 text-center text-slate-400">
            <i data-lucide="search-x" class="w-10 h-10 mx-auto text-slate-300 mb-3"></i>
            <h4 class="font-bold text-slate-700 text-base">No checkers found</h4>
            <p class="text-xs text-slate-500 mt-1">Try searching for something else like "WhatsApp", "Binance", "Revolut", or "Email".</p>
          </div>
        `;
        lucide.createIcons();
        return;
      }

      // Group by catTitle
      const groups = {};
      filtered.forEach(s => {
        if (!groups[s.catTitle]) groups[s.catTitle] = [];
        groups[s.catTitle].push(s);
      });

      let html = '';
      for (const [title, services] of Object.entries(groups)) {
        html += `
          <div class="bg-white rounded-2xl border border-slate-200/90 p-5 sm:p-7 shadow-xs">
            <div class="flex items-center justify-between pb-3.5 mb-4 border-b border-slate-100">
              <span class="text-xs font-bold uppercase tracking-wider text-slate-400 flex items-center gap-2">
                <span>${title}</span>
                <span class="px-2 py-0.5 rounded-full text-[10px] font-mono font-bold bg-slate-100 text-slate-600">${services.length}</span>
              </span>
            </div>
            <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-2.5">
              ${services.map(s => {
                const costVal = s.cost || 1;
                const costBadge = (costVal === 5)
                  ? `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-bold bg-amber-50 text-amber-700 border border-amber-200">5 cr</span>`
                  : (costVal === 3)
                  ? `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-bold bg-indigo-50 text-indigo-700 border border-indigo-200">3 cr</span>`
                  : (costVal === 2)
                  ? `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-semibold bg-blue-50 text-blue-700 border border-blue-200">2 cr</span>`
                  : `<span class="px-1.5 py-0.5 rounded text-[10px] font-mono font-medium bg-slate-100 text-slate-600 border border-slate-200">1 cr</span>`;

                const slug = s.name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '');
                return `
                <a href="/products/${slug}" class="bg-white hover:bg-blue-50/40 border border-slate-200 hover:border-[#435ebf] rounded-xl p-3 flex items-center justify-between transition cursor-pointer group shadow-2xs">
                  <div class="flex items-center gap-2.5 min-w-0 pr-2">
                    <div class="w-6 h-6 shrink-0 flex items-center justify-center">
                      ${getBrandSvg(s.icon, 20)}
                    </div>
                    <span class="text-xs sm:text-sm font-medium text-slate-800 group-hover:text-[#435ebf] transition truncate">${s.name}</span>
                  </div>
                  <div class="flex items-center gap-1.5 shrink-0">
                    ${costBadge}
                    ${renderBadge(s.badge)}
                  </div>
                </a>
              `;
              }).join('')}
            </div>
          </div>
        `;
      }

      container.innerHTML = html;
      lucide.createIcons();
    }

    async function runSpecificServiceCheck() {
      if (!activeModalService) return;
      const target = document.getElementById("modalTargetInput").value.trim();
      if (!target) return alert("Please enter a target phone number or email.");

      const btn = document.getElementById("btnRunServiceCheck");
      const orig = btn.innerHTML;
      btn.disabled = true;
      btn.innerHTML = `<i data-lucide="loader-2" class="w-4 h-4 animate-spin"></i><span>Executing Live Verification...</span>`;
      lucide.createIcons();

      try {
        const res = await fetch("/v1/verify/service", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            service_id: activeModalService.id,
            target: target,
            default_country: "MA",
            user_id: currentUser ? currentUser.id : null,
          })
        });

        if (res.status === 429) {
          closeServiceModal();
          openTrialLimitModal();
          fetchGuestQuota();
          return;
        }

        if (res.status === 402) {
          const errData = await res.json().catch(() => ({}));
          closeServiceModal();
          openTopUpModal('starter');
          return;
        }

        if (!res.ok) throw new Error("Service verification failed");
        const data = await res.json();
        displayServiceResult(data);

        if (currentUser) {
          refreshUserProfile();
          loadUserHistory();
        } else {
          fetchGuestQuota();
        }
      } catch (err) {
        console.error(err);
        alert(err.message);
      } finally {
        btn.disabled = false;
        btn.innerHTML = orig;
        lucide.createIcons();
      }
    }

          loadUserHistory();
        } else {
          fetchGuestQuota();

      } finally {
        btn.innerHTML = orig;
        lucide.createIcons();
      }
    }

    async function executeBatchVerification(numbers) {
      if (currentUser && currentUser.credits < numbers.length) {
        alert(`Insufficient credits! This batch requires ${numbers.length} credits, but your wallet has ${currentUser.credits}.`);
        return;
      }

      const res = await fetch("/v1/verify/batch", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          numbers: numbers,
          default_country: "MA",
          check_whatsapp: true,
          check_telegram: true,
          check_imessage: true,
          user_id: currentUser ? currentUser.id : null,
        })
      });

      if (res.status === 402) {
        alert("Insufficient credits! Please top up your wallet.");
        return;
      }

      if (res.ok) {
        const data = await res.json();
        bulkRecords = data.results;
        renderBulkTable(data.results);

        if (currentUser) {
          refreshUserProfile();
          loadUserHistory();
        }
      }
    }

    async function loadUserHistory() {
      if (!currentUser || !currentUser.id) return;

            t.lead_quality === "EXCELLENT" ? "bg-emerald-50 text-emerald-700 border-emerald-200" :
            t.lead_quality === "HIGH" ? "bg-blue-50 text-blue-700 border-blue-200" :

    function setupDragAndDrop() {
      const dropZone = document.getElementById("csvDropZone");
      if (!dropZone) return;

      ['dragenter', 'dragover'].forEach(eventName => {
        dropZone.addEventListener(eventName, (e) => {
          e.preventDefault();
          e.stopPropagation();
          dropZone.classList.add("border-[#435ebf]", "bg-blue-50/60");
        }, false);
      });

      ['dragleave', 'drop'].forEach(eventName => {
        dropZone.addEventListener(eventName, (e) => {
          e.preventDefault();
          e.stopPropagation();
          dropZone.classList.remove("border-[#435ebf]", "bg-blue-50/60");
        }, false);
      });

      dropZone.addEventListener('drop', (e) => {
        const dt = e.dataTransfer;
        const files = dt.files;
        if (files && files.length > 0) {
          handleBulkFileSelect(files[0]);
        }
      }, false);
    }

    function handleBulkFileSelect(file) {
      if (!file) return;
      if (!file.name.endsWith(".csv") && !file.name.endsWith(".txt")) {
        alert("Please select a .csv or .txt file.");
        return;
      }
      selectedBatchFile = file;

      // Read preview line count
      const reader = new FileReader();
      reader.onload = function(e) {
        const text = e.target.result;
        const lines = text.split(/\r?\n/).map(l => l.trim()).filter(Boolean);
        const count = Math.max(1, lines.length - (lines[0].toLowerCase().includes("phone") ? 1 : 0));
        const sizeKb = (file.size / 1024).toFixed(1);

        document.getElementById("batchFileName").innerText = file.name;
        document.getElementById("batchFileMeta").innerText = `${count.toLocaleString()} rows detected • ${sizeKb} KB`;

        document.getElementById("fileInfoBanner").classList.remove("hidden");
        document.getElementById("csvDropZone").classList.add("hidden");

    async function startBatchFileUpload() {
      if (!selectedBatchFile) {
        alert("Please drop or choose a file first.");
        return;
      }

      const btn = document.getElementById("btnStartBatchProcess");
      btn.disabled = true;
      btn.innerHTML = `<i data-lucide="loader-2" class="w-3.5 h-3.5 animate-spin"></i><span>Processing...</span>`;
      lucide.createIcons();

      const progressContainer = document.getElementById("batchProgressBarContainer");
      const progressBar = document.getElementById("batchProgressBar");
      const progressPercent = document.getElementById("batchProgressPercent");
      const progressStatus = document.getElementById("batchProgressStatus");

      progressContainer.classList.remove("hidden");
      progressBar.style.width = "20%";
      progressPercent.innerText = "20%";
      progressStatus.innerHTML = `<i data-lucide="loader-2" class="w-4 h-4 animate-spin text-[#435ebf]"></i> Parsing CSV rows...`;
      lucide.createIcons();

      const formData = new FormData();
      formData.append("file", selectedBatchFile);
      if (currentUser && currentUser.id) {
        formData.append("user_id", currentUser.id);
      }
      formData.append("default_country", "MA");
      formData.append("check_whatsapp", document.getElementById("checkOptWhatsApp").checked);
      formData.append("check_telegram", document.getElementById("checkOptTelegram").checked);
      formData.append("check_imessage", document.getElementById("checkOptApple").checked);

      setTimeout(() => {
        progressBar.style.width = "60%";
        progressPercent.innerText = "60%";
        progressStatus.innerHTML = `<i data-lucide="loader-2" class="w-4 h-4 animate-spin text-[#435ebf]"></i> Cascading WhatsApp, HLR, & Apple network...`;
        lucide.createIcons();
      }, 500);

      try {
        const res = await fetch("/v1/batch/upload-file", {
          method: "POST",
          body: formData,
        });

        if (res.status === 402) {
          alert("Insufficient wallet credits for this batch size! Please top up your credits.");
          progressBar.style.width = "0%";
          progressContainer.classList.add("hidden");
          return;
        }

        const data = await res.json();
        if (!res.ok) throw new Error(data.detail || "Batch verification failed");

        progressBar.style.width = "100%";
        progressPercent.innerText = "100%";
        progressStatus.innerHTML = `✅ Successfully enriched ${data.total_processed} contacts (${data.valid_count} valid, ${data.invalid_count} dead/invalid)`;

        lastBatchExportCsv = data.csv_export_content;
        bulkRecords = data.results || [];
        renderBulkTable(bulkRecords);

        if (currentUser) {
          refreshUserProfile();
          loadUserHistory();
        }

      } catch (err) {
        alert("Batch processing error: " + err.message);
      } finally {
        btn.disabled = false;
        btn.innerHTML = `<i data-lucide="play" class="w-3.5 h-3.5 fill-current"></i><span>Clean & Verify File Now</span>`;
        lucide.createIcons();
      }
    }

    function renderBulkTable(records) {
      const tbody = document.getElementById("bulkTableBody");
      if (!tbody) return;
      if (!records || records.length === 0) {
        tbody.innerHTML = `
          <tr>
            <td colspan="9" class="p-8 text-center text-slate-400 font-sans">
              No contacts found to display.
            </td>
          </tr>
        `;
        return;
      }
      tbody.innerHTML = records.map(r => {
        const qualityBadge = 
          r.lead_quality === "EXCELLENT" ? "bg-emerald-50 text-emerald-700 border-emerald-200" :
          r.lead_quality === "HIGH" ? "bg-blue-50 text-blue-700 border-blue-200" :
          r.lead_quality === "LOW" ? "bg-amber-50 text-amber-700 border-amber-200" :
          "bg-rose-50 text-rose-700 border-rose-200";

        const country = r.phone ? (r.phone.country_iso || "GLOBAL") : "GLOBAL";
        const carrierName = r.carrier ? (r.carrier.carrier_name || "—") : "—";
        const isVoip = r.carrier ? r.carrier.is_voip : false;

        return `
          <tr class="hover:bg-slate-50 transition">
            <td class="p-3 text-slate-500">${r.raw_input || '—'}</td>
            <td class="p-3 font-bold text-slate-900">${(r.phone && r.phone.e164) ? r.phone.e164 : '—'}</td>
            <td class="p-3 font-semibold text-slate-700 font-sans">${country}</td>
            <td class="p-3">
              <span class="text-slate-800 font-medium block truncate max-w-[130px]">${carrierName}</span>
              <span class="text-[10px] text-slate-400 font-sans">${r.phone ? (r.phone.line_type || '') : ''}</span>
            </td>
            <td class="p-3">
              ${isVoip ? '<span class="text-amber-600 font-bold">VOIP</span>' : '<span class="text-slate-400">Mobile</span>'}
            </td>
            <td class="p-3">
              ${r.whatsapp && r.whatsapp.is_registered ? '<span class="text-emerald-600 font-bold">Active</span>' : '<span class="text-slate-400">None</span>'}
            </td>
            <td class="p-3">
              ${r.imessage && r.imessage.is_registered ? '<span class="text-[#435ebf] font-bold">Apple</span>' : '<span class="text-slate-400">None</span>'}
            </td>
            <td class="p-3 font-bold text-slate-800">${r.reachability_score}</td>
            <td class="p-3">
              <span class="px-2 py-0.5 rounded-full text-[10px] font-bold border ${qualityBadge}">
                ${r.lead_quality}
              </span>
            </td>
          </tr>
        `;
      }).join("");
    }

    function downloadEnrichedBatchCsv() {
      if (lastBatchExportCsv) {
        const blob = new Blob([lastBatchExportCsv], { type: "text/csv;charset=utf-8;" });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = `pulsecheck_cleaned_${Date.now()}.csv`;
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
      } else if (bulkRecords && bulkRecords.length > 0) {
        downloadCleanedCsv();
      } else {
        alert("No batch results available yet. Please upload a file or run the sample demo first.");
      }
    }

    function downloadCleanedCsv() {
      if (!bulkRecords || bulkRecords.length === 0) return alert("Please load or upload a batch list first.");

      const headers = ["Raw Input", "Normalized E164", "Country", "Carrier", "Line Type", "Is VoIP", "WhatsApp Active", "Telegram Active", "iMessage Active", "Reachability Score", "Lead Quality"];
      const rows = bulkRecords.map(r => [
        `"${r.raw_input || ''}"`,
        `"${r.phone ? r.phone.e164 || '' : ''}"`,
        `"${r.phone ? r.phone.country_iso || '' : ''}"`,
        `"${r.carrier ? r.carrier.carrier_name || '' : ''}"`,
        `"${r.phone ? r.phone.line_type || '' : ''}"`,
        `"${r.carrier ? (r.carrier.is_voip ? 'YES' : 'NO') : 'NO'}"`,
        `"${r.whatsapp ? (r.whatsapp.is_registered ? 'ACTIVE' : 'INACTIVE') : ''}"`,
        `"${r.telegram ? (r.telegram.is_registered ? 'ACTIVE' : 'INACTIVE') : ''}"`,
        `"${r.imessage ? (r.imessage.is_registered ? 'ACTIVE' : 'INACTIVE') : ''}"`,
        `"${r.reachability_score}"`,
        `"${r.lead_quality}"`
      ]);

      const csv = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(e => e.join(","))].join("\n");
      const a = document.createElement("a");
      a.href = encodeURI(csv);
      a.download = `pulsecheck_clean_list_${Date.now()}.csv`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
    }
